cmake_minimum_required(VERSION 3.16)
project(OrderMatchingEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(orderbook src/OrderBook.cpp)
target_include_directories(orderbook PUBLIC include)
if (MSVC)
  target_compile_options(orderbook PRIVATE /W4 /O2)
else()
  target_compile_options(orderbook PRIVATE -Wall -Wextra -Wpedantic -O3)
endif()
add_executable(matching_engine src/main.cpp)
target_link_libraries(matching_engine PRIVATE orderbook)
add_executable(orderbook_tests tests/test_order_book.cpp)
target_link_libraries(orderbook_tests PRIVATE orderbook)
add_executable(orderbook_benchmark benchmarks/benchmark.cpp)
target_link_libraries(orderbook_benchmark PRIVATE orderbook)
enable_testing()
add_test(NAME orderbook_tests COMMAND orderbook_tests)
